```js
/** A barra de progresso gruda logo abaixo do cabeçalho público quando ele é fixo (a partir de 760 px). */
function medirTopo(alvo) {
  const cabecalho = document.querySelector('.pub-barra');
  const fixo = cabecalho && getComputedStyle(cabecalho).position === 'sticky';
  const altura = fixo ? Math.round(cabecalho.getBoundingClientRect().height) : 0;
  alvo.style.setProperty('--laift-progresso-topo', `${altura}px`);
}

function iniciarProgresso(artigo) {
  const alvo = document.querySelector('[data-laift-progresso]');
  const LP = window.LaiftProgress;
  if (!alvo || !LP) return;
  const controle = LP.montarLeitura({ alvo, conteudo: artigo, id: alvo.getAttribute('data-laift-progresso'), sprite: SPRITE });
  if (!controle) return;
  medirTopo(alvo);
  window.addEventListener('resize', () => medirTopo(alvo), { passive: true });
}
```
